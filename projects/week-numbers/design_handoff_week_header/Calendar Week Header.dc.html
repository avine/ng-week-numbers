export class WeekHeader<D> {
  calendar = inject(MatCalendar<D>);
  adapter  = inject(DateAdapter<D>);
  constructor(cdr: ChangeDetectorRef) {
    this.calendar.stateChanges.pipe(takeUntilDestroyed()).subscribe(() => cdr.markForCheck());
  }
  // semaines ISO de chaque ligne du mois de calendar.activeDate
  get weeks() { /* 1er du mois → lundi de sa semaine → pas de 7 j jusqu'à la fin du mois */ }
  goTo(monday: D) { this.calendar.activeDate = monday; }
}

En vue année / multi-années, masquer le bloc semaines (calendar.currentView !== 'month').
